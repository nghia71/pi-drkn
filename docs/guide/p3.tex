%======================================================== PB
\begin{frame}[label=pb]{\rtag{PB}\ Phản biện}
\rolecols{PB}{%
• Chỉ \textbf{bài được giao cho bạn} trong kỳ đang mở.\\[1mm]
• Đề bài và lời giải đã biên tập.\\[1mm]
• Thảo luận của bài đó (cả của phản biện khác cùng được giao).\\[3mm]
\textbf{Không thấy}: tên tác giả, tên tệp, nguồn, các bài khác — để chấm \textbf{ẩn danh}.\\[3mm]
Hết kỳ, bài rời khỏi danh sách của bạn.}{%
\textbf{1.}\ Nhận email mời, mở \textbf{đường dẫn đăng nhập}.\\[1mm]
\textbf{2.}\ Mở từng bài: tự giải, rồi đọc lời giải đi kèm.\\[1mm]
\textbf{3.}\ Viết nhận xét ngay dưới bài (mục \emph{Thảo luận}). Nên trả lời:\\[0.5mm]
\hspace*{3mm}• Đề có rõ, đủ điều kiện không?\\
\hspace*{3mm}• Lời giải đúng và đủ chưa? Chỗ nào cần sửa?\\
\hspace*{3mm}• Mức đề nghị: \textbf{A} hay \textbf{B}?\\
\hspace*{3mm}• Có giống bài đã biết không?\\
\hspace*{3mm}• Đề nghị: \textbf{chọn} / \textbf{sửa rồi chọn} / \textbf{không chọn}.\\[1mm]
\textbf{4.}\ Đọc và trả lời nhận xét của người khác trên cùng bài.\\[2mm]
\textcolor{ink!65}{Công thức viết như \LaTeX: \texttt{\$a\textbackslash ge b\$} (mục 7). Nhận xét chưa sửa được sau khi gửi.}}{%
• Ô chấm riêng: mức đề nghị, đề nghị chọn / không.\\[1mm]
• Hạn phản biện hiện trên trang.\\[1mm]
• Email nhắc trước hạn.\\[1mm]
• Đánh dấu \emph{đã xong} từng bài.}
\end{frame}

%======================================================== NCB
\begin{frame}[label=ncb]{\rtag{NCB}\ Người chuẩn bị bài}
\rolecols{NCB}{%
• \textbf{Mọi bài}, tên tác giả (không có liên hệ).\\[1mm]
• \textbf{Nguồn \& chỉnh sửa} của từng bài (trang sau).\\[1mm]
• Chấm \textbf{cần kiểm tra n}, \textbf{xung đột mở n} trên danh sách.\\[1mm]
• Thảo luận của mọi bài.}{%
\textbf{Nguyên tắc}\\[0.5mm]
• \textbf{Không sửa lén}: mỗi sửa đổi là một dòng ghi — vị trí, trước, sau, lý do, trạng thái. Văn bản đề, lời giải không chứa ghi chú.\\
• \textbf{Bản gốc của tác giả giữ nguyên}; chỉ sửa bản biên tập.\\
• \textbf{Nghi ngờ thì ghi, không tự sửa}: thành mục \emph{cần kiểm tra}; khi cần, hỏi tác giả qua Văn phòng / PT.\\
• Xung đột ghi theo loại, mỗi cái một dòng.\\[2mm]
\textbf{Bây giờ}\\[0.5mm]
• Lọc danh sách, xử lý các bài có \emph{cần kiểm tra}.\\
• Đối chiếu bản gốc khi mục \emph{Đối chiếu nguyên bản} còn „chưa".\\
• Trả lời thảo luận; báo dòng \emph{Cảnh báo hiển thị}.\\
• \textbf{Sửa đề, lời giải}: nút \textbf{Sửa} cạnh tiêu đề; bên phải xem trước công thức; bấm \textbf{Lưu}. Người khác vừa lưu thì hệ thống báo, bản bạn gõ vẫn còn — không ai bị ghi đè.\\
• \textbf{Lịch sử sửa} (cuối mục Nguồn \& chỉnh sửa): ai sửa, khi nào, chữ bỏ gạch đỏ, chữ thêm tô xanh.}{%
• Thêm / đóng mục cần kiểm tra, xung đột.\\[1mm]
• Biểu mẫu \textbf{Nhập bài}: tải hồ sơ, hệ thống tạo mã bài.}
\end{frame}

%======================================================== PT
\begin{frame}[label=pt]{\rtag{PT}\ Phụ trách chuyên mục}
\rolecols{PT}{%
• \textbf{Mọi bài}, tên và \textbf{liên hệ} tác giả.\\[1mm]
• Nguồn \& chỉnh sửa, xung đột, mục cần kiểm tra.\\[1mm]
• Thảo luận của mọi bài.}{%
• Theo dõi \textbf{việc còn mở}: lọc theo trạng thái, xem các chấm \emph{cần kiểm tra}, \emph{xung đột mở}.\\[1mm]
• Giải quyết xung đột trong phạm vi PT (số hiệu, bản chép khác nhau, trùng bài…); xung đột \emph{mức}, \emph{tác giả} chuyển TBT quyết định.\\[1mm]
• Liên hệ tác giả khi NCB cần hỏi (thông tin liên hệ chỉ dùng trong hệ thống, không chép ra đề, lời giải).\\[1mm]
• Nhận người mới: báo Quản trị email và vai trò của họ.\\[1mm]
• Sửa đề, lời giải ngay trên trang (như NCB).\\[1mm]
• Đổi trạng thái bài (Mới → SL → SL-OK / SL-Fail → PL): hiện báo Quản trị, cho tới khi có nút.}{%
• Nút \textbf{đổi trạng thái}, thêm xung đột.\\[1mm]
• \textbf{Mở kỳ phản biện}: chọn bài, giao cho phản biện, đặt hạn; email mời tự gửi.\\[1mm]
• \textbf{Bảng chọn bài}: xếp bài theo mức, vị trí; TBT duyệt.\\[1mm]
• \textbf{Khoá kỳ}: đánh số in, xuất \LaTeX\ cho BTK.}
\end{frame}

• Thảo luận của mọi bài, gồm nhận xét của phản biện.}{%

• \textbf{Mức} A / B khi tác giả, NCB và phản biện khác nhau.\\
• \textbf{Tác giả}: sáng tác hay sưu tầm, cách ghi tên khi đăng.\\
• Bài nghi \textbf{trùng} bài đã biết: giữ, sửa hay bỏ.\\
• \textbf{Duyệt cuối} danh sách đăng.\\[2mm]
\textbf{Bây giờ}\\[0.5mm]
• Lọc danh sách; mở bài có \emph{xung đột mở} để xem loại xung đột.\\
• Ghi quyết định vào Thảo luận của bài — mọi người cùng thấy, không cần email.\\
• Muốn xem hệ thống như phản biện thấy: nhờ Quản trị dùng \emph{xem như vai trò}.}{%
• Nút ghi quyết định cho xung đột, đổi trạng thái.\\[1mm]
• \textbf{Bảng chọn bài}: xem đề xuất của PT, duyệt.\\[1mm]
• Thông báo email gọn: bài nào chờ TBT quyết định.}
\end{frame}
